package caixaEletronico;

import java.util.Scanner;
import java.util.Locale;


public class Principal {
	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		boolean saida = false;
		float saldo = 0;
		float limite = 0;
		int senha = 123456;
		int lime = 0;
		while (lime<=2) {
			System.out.println("Digite sua senha");
			int tentativa = sc.nextInt();
			if (tentativa != 123456) {
				lime++;
				
				if (lime <= 2) {
					System.out.println("Senha invalida! tente novamente");
					
				}
				
				if (lime == 3) {
					System.out.println("Conta bloqueada pequeno porco!");
				}
			} else {
				while (!saida) {
			    	System.out.println(":-------------------- \n");
			    	System.out.println("|Digite 1 para Consultar saldo \n");
			        System.out.println("|Digite 2 para Despositar\n");
			    	System.out.println("|Digite 3 para Sacar\n");
			    	System.out.println("|Digite 4 para Sair \n");
			    	System.out.println(":-------------------- \n");
			    	int opcao = sc.nextInt();
			    	if (opcao==1) {
			    		System.out.printf(new Locale("pt", "BR"), "O valor na conta é R$ %.2f%n", saldo);
			    	}
			    	else if(opcao==2) {
				    	System.out.println("Informe quanto quer depositar (em reais): ");
				    	float deposito = sc.nextFloat();
				    	if (deposito<=0) {
					    	System.out.println("Valor digitado não é possível!");
				    	}
				    	else {
				    		saldo+=deposito;
					    	System.out.println("Depósito realizado com sucesso!");
				    	}
				    	
			    	} else if (opcao==3) {
			    		System.out.println("Informe o valor que deseja sacar (em reais): ");
			    		float saque = sc.nextFloat();
			    		if (saque<=0) {
			    			System.out.println("Valor digitado não é possível!");	
			    		} else if (saque>saldo) {
			    			System.out.println("Seu saldo é insuficiente!");
			    		} else {
			    			if ((limite<=500) && ((saque+limite)<= 500)) {
			    				saldo-=saque;
			    				System.out.println("Saque realizado com sucesso!");
			    				limite+=saque;
			    			}else {
			    				System.out.println("O seu limite diário já foi consumido!");
			    			}
			    		}
			    	} 
			    	else if (opcao==4) {
		    			System.out.println("Sessão finalizada!");
		    			saida = true;
			    	}
			    	else {
		    			System.out.println("Opção digitada inválida!");
			    	}
				}
				sc.close();
			}
		} 
		
	}
}
