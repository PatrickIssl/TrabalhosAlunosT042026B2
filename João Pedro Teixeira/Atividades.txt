import java.util.Scanner;

class ContaBancaria {
    String titular;
    double saldo;

    void depositar(double valor) {
        if (valor > 0) {
            saldo = saldo + valor;
            System.out.println("Deposito realizado com sucesso.");
        } else {
            System.out.println("O valor do deposito deve ser positivo.");
        }
    }

    void sacar(double valor) {
        if (valor > 0 && valor <= saldo) {
            saldo = saldo - valor;
            System.out.println("Saque realizado com sucesso.");
        } else {
            System.out.println("Saque nao pode ser realizado: saldo insuficiente ou valor invalido.");
        }
    }

    void exibirSaldo() {
        System.out.printf("Saldo atual: R$ %.2f%n", saldo);
    }
}

public class Atividade05_ContaBancaria {
    public static void main(String[] args) {
        Scanner entrada = new Scanner(System.in);
        ContaBancaria conta = new ContaBancaria();

        System.out.print("Digite o nome do titular: ");
        conta.titular = entrada.nextLine();

        System.out.print("Digite o saldo inicial: R$ ");
        conta.saldo = entrada.nextDouble();

        System.out.print("Digite o valor do deposito: R$ ");
        conta.depositar(entrada.nextDouble());

        System.out.print("Digite o valor do saque: R$ ");
        conta.sacar(entrada.nextDouble());

        System.out.println("Titular: " + conta.titular);
        conta.exibirSaldo();
        entrada.close();
    }
}

import java.util.Scanner;

class Livro {
    String titulo;
    String autor;
    int numeroPaginas;
    boolean emprestado = false;

    void emprestar() {
        if (!emprestado) {
            emprestado = true;
            System.out.println("Livro emprestado com sucesso.");
        } else {
            System.out.println("O livro ja esta emprestado.");
        }
    }

    void devolver() {
        if (emprestado) {
            emprestado = false;
            System.out.println("Livro devolvido com sucesso.");
        } else {
            System.out.println("O livro nao esta emprestado.");
        }
    }

    void exibirInformacoes() {
        System.out.println("Titulo: " + titulo);
        System.out.println("Autor: " + autor);
        System.out.println("Numero de paginas: " + numeroPaginas);
        System.out.println("Situacao: " + (emprestado ? "Emprestado" : "Disponivel"));
    }
}

public class Atividade06_Livro {
    public static void main(String[] args) {
        Scanner entrada = new Scanner(System.in);
        Livro livro = new Livro();

        System.out.print("Digite o titulo: ");
        livro.titulo = entrada.nextLine();

        System.out.print("Digite o autor: ");
        livro.autor = entrada.nextLine();

        System.out.print("Digite o numero de paginas: ");
        livro.numeroPaginas = entrada.nextInt();

        livro.exibirInformacoes();
        livro.emprestar();
        livro.emprestar();
        livro.exibirInformacoes();
        livro.devolver();
        livro.devolver();
        livro.exibirInformacoes();

        entrada.close();
    }
}

import java.util.Scanner;

class Funcionario {
    String nome;
    String cargo;
    double salario;

    void aumentarSalario(double percentual) {
        if (percentual > 0) {
            salario = salario + (salario * percentual / 100);
        } else {
            System.out.println("O percentual deve ser positivo.");
        }
    }
}

public class Atividade07_Funcionario {
    public static void main(String[] args) {
        Scanner entrada = new Scanner(System.in);
        Funcionario funcionario = new Funcionario();

        System.out.print("Digite o nome: ");
        funcionario.nome = entrada.nextLine();

        System.out.print("Digite o cargo: ");
        funcionario.cargo = entrada.nextLine();

        System.out.print("Digite o salario: R$ ");
        funcionario.salario = entrada.nextDouble();

        System.out.print("Digite o percentual de aumento: ");
        double percentual = entrada.nextDouble();

        funcionario.aumentarSalario(percentual);

        System.out.println("Nome: " + funcionario.nome);
        System.out.println("Cargo: " + funcionario.cargo);
        System.out.printf("Novo salario: R$ %.2f%n", funcionario.salario);
        entrada.close();
    }
}

import java.util.Scanner;

class Retangulo {
    double largura;
    double altura;

    double calcularArea() {
        return largura * altura;
    }

    double calcularPerimetro() {
        return 2 * (largura + altura);
    }
}

public class Atividade08_Retangulo {
    public static void main(String[] args) {
        Scanner entrada = new Scanner(System.in);
        Retangulo retangulo = new Retangulo();

        System.out.print("Digite a largura: ");
        retangulo.largura = entrada.nextDouble();

        System.out.print("Digite a altura: ");
        retangulo.altura = entrada.nextDouble();

        System.out.printf("Area: %.2f%n", retangulo.calcularArea());
        System.out.printf("Perimetro: %.2f%n", retangulo.calcularPerimetro());
        entrada.close();
    }
}

    void carregar() {
        bateria = 100;

import java.util.Scanner;

class PessoaCadastro {
    String nome;
    int idade;
    String cidade;

    void apresentar() {
        System.out.println("Ola, meu nome e " + nome + " e tenho " + idade + " anos.");
    }

    void fazerAniversario() {
        idade = idade + 1;
    }

    void exibirDados() {
        System.out.println("Nome: " + nome);
        System.out.println("Idade: " + idade);
        System.out.println("Cidade: " + cidade);
    }
}

public class Atividade10_CadastroPessoas {
    static void cadastrarPessoa(Scanner entrada, PessoaCadastro pessoa, int numero) {
        System.out.println("\nCadastro da pessoa " + numero);

        System.out.print("Nome: ");
        pessoa.nome = entrada.nextLine();

        System.out.print("Idade: ");
        pessoa.idade = Integer.parseInt(entrada.nextLine());

        System.out.print("Cidade: ");
        pessoa.cidade = entrada.nextLine();
    }

    public static void main(String[] args) {
        Scanner entrada = new Scanner(System.in);

        PessoaCadastro pessoa1 = new PessoaCadastro();
        PessoaCadastro pessoa2 = new PessoaCadastro();
        PessoaCadastro pessoa3 = new PessoaCadastro();

        cadastrarPessoa(entrada, pessoa1, 1);
        cadastrarPessoa(entrada, pessoa2, 2);
        cadastrarPessoa(entrada, pessoa3, 3);

        System.out.println("\n--- Dados das pessoas ---");
        pessoa1.apresentar();
        pessoa1.exibirDados();

        System.out.println();
        pessoa2.apresentar();
        pessoa2.exibirDados();

        System.out.println();
        pessoa3.apresentar();
        pessoa3.exibirDados();

        System.out.println("\nA primeira pessoa fez aniversario!");
        pessoa1.fazerAniversario();

        System.out.println("\n--- Dados atualizados ---");
        pessoa1.exibirDados();
        System.out.println();
        pessoa2.exibirDados();
        System.out.println();
        pessoa3.exibirDados();

        entrada.close();
    }
}
